    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const lightbox = document.getElementById('lightbox');
            const lightboxImg = document.getElementById('lightbox-img');
            const closeBtn = document.querySelector('.close-lightbox');
            const prevBtn = document.getElementById('prev-btn');
            const nextBtn = document.getElementById('next-btn');
            
            let currentImages = [];
            let currentIndex = 0;

            // Находим все карточки и вешаем события на их картинки отдельно
            document.querySelectorAll('.card').forEach(card => {
                // Собираем все картинки внутри конкретной карточки (и главную, и галерейные)
                const cardImgs = Array.from(card.querySelectorAll('.card-main-img, .photo-grid img'));
                
                cardImgs.forEach(img => {
                    img.addEventListener('click', (e) => {
                        // Останавливаем сворачивание details и открытие стандартных событий
                        e.preventDefault();
                        e.stopPropagation();
                        
                        // Запоминаем массив картинок именно этой услуги
                        currentImages = cardImgs.map(i => i.src);
                        currentIndex = currentImages.indexOf(img.src);
                        
                        updateLightboxImage();
                        lightbox.classList.add('active');
                    });
                });
            });

            function updateLightboxImage() {
                lightboxImg.src = currentImages[currentIndex];
            }

            // Кнопка Назад
            prevBtn.addEventListener('click', (e) => {
                e.stopPropagation();
                currentIndex = (currentIndex - 1 + currentImages.length) % currentImages.length;
                updateLightboxImage();
            });

            // Кнопка Вперед
            nextBtn.addEventListener('click', (e) => {
                e.stopPropagation();
                currentIndex = (currentIndex + 1) % currentImages.length;
                updateLightboxImage();
            });

            // Закрытие на крестик
            closeBtn.addEventListener('click', () => {
                lightbox.classList.remove('active');
            });

            // Закрытие по клику на фон вне картинки
            lightbox.addEventListener('click', (e) => {
                if (e.target === lightbox) {
                    lightbox.classList.remove('active');
                }
            });

            // Навигация с клавиатуры (стрелочки и Esc)
            document.addEventListener('keydown', (e) => {
                if (!lightbox.classList.contains('active')) return;
                if (e.key === 'Escape') lightbox.classList.remove('active');
                if (e.key === 'ArrowLeft') prevBtn.click();
                if (e.key === 'ArrowRight') nextBtn.click();
            });
        });
    </script>
